Snippet 1 - Saludo con funcion

def saludar(nombre, edad=None, carrera=None): mensaje = f"Hola {nombre}" if edad is not None: if edad < 18: mensaje += f"\nQue joven eres con {edad} anios" else: mensaje += f"\nToda una experiencia con {edad} anios" if carrera is not None: mensaje += f"\nEstudias {carrera}, mucho exito" return mensaje print(saludar("Pepe")) print("---") print(saludar("Juan", 21, "Informatica"))

Snippet 3 - Operaciones matematicas

def sumar(x, y): r = x + y if r > 100: print("El resultado es demasiado grande") return r def dividir(x, y): if y == 0: return "No se puede dividir por cero" return round(x / y, 2) print("sumar(13, 9) =", sumar(13, 9)) print("dividir(11, 5) =", dividir(11, 5))
